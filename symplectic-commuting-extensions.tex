\title{Extending commuting symplectic embeddings\\
to commuting symplectic automorphisms}
\author{}
\date{}

\begin{document}
\maketitle

\begin{abstract}
Let \(H=\ell^2(\N;\R^{2d})=\bigoplus_{n\ge1}\R^{2d}\) with symplectic form
\(\Omega=\bigoplus_n \omega_0\) given by \(\cJ=\bigoplus_n J_{2d}\).
We prove, line by line:
\begin{enumerate}
\item every bounded \(T\) with \(T^*\cJ T=\cJ\) extends to a symplectic automorphism
of a larger symplectic Hilbert space;
\item if \(T,T'\) are commuting maps of this type, they admit commuting symplectic
automorphism extensions on a common enlargement.
\end{enumerate}
\end{abstract}

\section{Setup}
%----------------------------------------------------------------------

Let \(V\) be a real separable Hilbert space with inner product \(g\) and norm
\(\|\cdot\|\). Fix a bounded operator \(\cJ\) on \(V\) satisfying
\begin{equation}
\label{eq:J}
\cJ^2=-I,\qquad \cJ^*=-\cJ.
\end{equation}
Define
\begin{equation}
\label{eq:Omega}
\Omega(x,y):=g(\cJ x,\,y).
\end{equation}
Then \(\Omega\) is a \emph{strong} symplectic form: \(x\mapsto\Omega(x,\cdot)\) is a
topological isomorphism \(V\to V^*\).

The motivating example is
\[
H=\ell^2(\N;\R^{2d})=\bigoplus_{n\ge1}\R^{2d},
\qquad
\cJ=\bigoplus_{n\ge1}J_{2d},
\qquad
\Omega=\bigoplus_{n\ge1}\omega_0.
\]

\begin{definition}
A bounded linear map \(T:V\to V\) is a \emph{symplectic embedding} if
\begin{equation}
\label{eq:emb}
\Omega(Tx,Ty)=\Omega(x,y)\quad\text{for all }x,y\in V,
\end{equation}
equivalently \(T^*\cJ T=\cJ\). It is a \emph{symplectic automorphism} if it is a
bijective symplectic embedding; then \(T^{-1}\) is automatically bounded and
\(T\in\mathrm{Sp}(V,\Omega)\).
\end{definition}

\begin{lemma}[Basic facts]
\label{lem:basic}
\begin{enumerate}[label=\textup{(\roman*)}]
\item \(\|\cJ x\|=\|x\|\) for all \(x\).
\item \eqref{eq:emb} \(\Longleftrightarrow\) \(T^*\cJ T=\cJ\).
\item \(\Omega\) is strong.
\end{enumerate}
\end{lemma}

\begin{proof}
(i) From \eqref{eq:J}, \(\cJ^*\cJ=I\), so \(\|\cJ x\|^2=g(x,\cJ^*\cJ x)=\|x\|^2\).

(ii) \(\Omega(Tx,Ty)=g(T^*\cJ Tx,y)\) and \(\Omega(x,y)=g(\cJ x,y)\).

(iii) The map \(\flat(x)=\Omega(x,\cdot)\) equals \(\iota_g\circ\cJ\), a composition of
isomorphisms.
\end{proof}

\section{Left inverse and closed range}
%----------------------------------------------------------------------

\begin{lemma}
\label{lem:left}
Let \(T\) be a symplectic embedding. Then:
\begin{enumerate}[label=\textup{(\roman*)}]
\item \(\|Tx\|\ge\|x\|/\|T\|\) for all \(x\) (so \(T\) is injective);
\item \(R:=\operatorname{ran}T\) is closed;
\item \(L:=-\cJ\,T^*\cJ\) satisfies \(LT=I\) and \(\|L\|\le\|T\|\);
\item \(E:=TL\) is a bounded projection onto \(R\).
\end{enumerate}
\end{lemma}

\begin{proof}
(i) \(\|x\|=\|\cJ x\|=\|T^*\cJ Tx\|\le\|T\|\,\|Tx\|\).

(ii) If \(Tx_n\to z\), then \((Tx_n)\) is Cauchy, hence so is \((x_n)\) by (i).
Thus \(x_n\to x\) and \(Tx=z\), so \(z\in R\).

(iii) \(LT=(-\cJ T^*\cJ)T=-\cJ(T^*\cJ T)=-\cJ^2=I\), and
\(\|L\|\le\|T\|\).

(iv) \(E^2=TLTL=T(LT)L=TL=E\) and \(\operatorname{ran}E=R\).
\end{proof}

\section{Symplectic complement of the range}
%----------------------------------------------------------------------

\begin{lemma}
\label{lem:N}
Let \(T\) be a symplectic embedding, \(R=\operatorname{ran}T\), and
\[
N:=R^{\perp_\Omega}
=\{v\in V:\Omega(v,r)=0\text{ for all }r\in R\}.
\]
Then:
\begin{enumerate}[label=\textup{(\roman*)}]
\item \(N=\cJ^{-1}(R^{\perp_g})\) is closed;
\item \(V=R\oplus N\) as a topological direct sum, and \(\Omega(R,N)=0\);
\item \((R,\Omega|_R)\) and \((N,\Omega|_N)\) are strong symplectic;
\item \(T:V\to R\) is a symplectic isomorphism;
\item \(T\) is surjective if and only if \(N=\{0\}\).
\end{enumerate}
\end{lemma}

\begin{proof}
(i) \(\Omega(v,r)=g(\cJ v,r)\).

(ii) Let \(P_R\) be the orthogonal projection onto \(R\) and
\(B:=P_R\cJ|_R:R\to R\). Then \(g(Br,s)=\Omega(r,s)\) for \(r,s\in R\), so
\(B\) is invertible (because \(\Omega|_R\) is strong by (iii) below).
Set \(\widetilde E:=B^{-1}P_R\cJ\). This is a bounded idempotent with
\(\operatorname{ran}\widetilde E=R\) and \(\ker\widetilde E=N\). Hence
\(V=R\oplus N\) topologically. (In fact \(\widetilde E=E=TL\).)

(iii) If \(n\in N\) and \(\Omega(n,N)=0\), then also \(\Omega(n,R)=0\), so
\(\Omega(n,V)=0\), hence \(n=0\). Given \(\phi\in N^*\), set
\(\tilde\phi=\phi\circ(I-E)\in V^*\). Choose \(x=r+n\) with \(\flat(x)=\tilde\phi\).
Then \(\phi(m)=\Omega(n,m)\) for \(m\in N\), so \(\flat_N\) is surjective, hence an
isomorphism. The same holds for \(R\).

(iv) Follows from Lemma~\ref{lem:left} and \eqref{eq:emb}.

(v) Clear.
\end{proof}

\section{Theorem A: single embedding}
%----------------------------------------------------------------------

\begin{theorem}[Single embedding]
\label{thm:A}
Let \(T:V\to V\) be a symplectic embedding. Then there exist a symplectic Hilbert space
\((W,\widehat\Omega)\) containing \((V,\Omega)\) as a closed symplectic subspace, and a
symplectic automorphism \(\widetilde T:W\to W\), such that \(\widetilde T|_V=T\).
\end{theorem}

\begin{proof}
If \(N=\{0\}\), then \(T\) is already an automorphism. Assume \(N\neq\{0\}\).

\medskip
\noindent\textbf{Step 1 (larger space).}
Take Hilbert copies \(N_k\) (\(k\ge1\)) of \(N\) with symplectic isometries
\(\iota_k:N\to N_k\). Set
\begin{equation}
\label{eq:W}
W:=V\oplus\bigoplus_{k\ge1}N_k
=\Bigl\{(v;n_1,n_2,\dots):\ \|v\|^2+\sum_k\|n_k\|^2<\infty\Bigr\},
\end{equation}
with the orthogonal-sum inner product, and
\begin{equation}
\label{eq:Omegahat}
\widehat\Omega\bigl((v;n_\bullet),(v';n'_\bullet)\bigr)
:=\Omega(v,v')+\sum_{k\ge1}\Omega(n_k,n'_k).
\end{equation}
Then \(W\) is complete (as an \(\ell^2\)-sum of Hilbert spaces). The form
\(\widehat\Omega\) is bounded by Cauchy--Schwarz, and
\(\widehat\flat=\flat\oplus\bigoplus_k\flat_N\) is an isomorphism, so
\(\widehat\Omega\) is strong. The embedding \(v\mapsto(v;0,0,\dots)\) realizes \(V\) as a
closed symplectic subspace of \(W\).

\medskip
\noindent\textbf{Step 2 (definition of \(\widetilde T\)).}
Define
\begin{equation}
\label{eq:Ttil}
\widetilde T(v;n_1,n_2,n_3,\dots)
=\bigl(Tv+\iota_1^{-1}n_1;\
\iota_1\iota_2^{-1}n_2,\
\iota_2\iota_3^{-1}n_3,\ \dots\bigr).
\end{equation}

\medskip
\noindent\textbf{Step 3 (boundedness).}
Each \(\iota_k\iota_{k+1}^{-1}\) is an isometry, so
\begin{align*}
\|\widetilde Tu\|^2
&=\|Tv+\iota_1^{-1}n_1\|^2+\sum_{k\ge1}\|n_{k+1}\|^2\\
&\le 2\|T\|^2\|v\|^2+2\|n_1\|^2+\sum_{k\ge1}\|n_{k+1}\|^2
\le C\|u\|^2.
\end{align*}

\medskip
\noindent\textbf{Step 4 (extends \(T\)).}
If \(n_\bullet=0\), then \(\widetilde T(v;0)=(Tv;0,\dots)\).

\medskip
\noindent\textbf{Step 5 (preserves \(\widehat\Omega\)).}
Write \(u=(v;n_\bullet)\), \(u'=(v';n'_\bullet)\). Then
\begin{align*}
\widehat\Omega(\widetilde Tu,\widetilde Tu')
&=\Omega\bigl(Tv+\iota_1^{-1}n_1,\,Tv'+\iota_1^{-1}n'_1\bigr)\\
&\qquad+\sum_{k\ge1}\Omega(n_{k+1},n'_{k+1}).
\end{align*}
Since \(Tv\in R\), \(\iota_1^{-1}n_1\in N\), and \(\Omega(R,N)=0\),
\begin{align*}
\Omega(Tv,Tv')&=\Omega(v,v'),\\
\Omega(Tv,\iota_1^{-1}n'_1)&=0=\Omega(\iota_1^{-1}n_1,Tv'),\\
\Omega(\iota_1^{-1}n_1,\iota_1^{-1}n'_1)&=\Omega(n_1,n'_1).
\end{align*}
Adding the shifted sum yields \(\widehat\Omega(\widetilde Tu,\widetilde Tu')=\widehat\Omega(u,u')\).

\medskip
\noindent\textbf{Step 6 (injectivity).}
If \(\widetilde Tu=0\), then \(\widehat\Omega(u,u')=0\) for all \(u'\), so \(u=0\).

\medskip
\noindent\textbf{Step 7 (surjectivity).}
Let \(w=(z;m_1,m_2,\dots)\in W\). Set
\(n_{k+1}:=\iota_{k+1}\iota_k^{-1}m_k\) for \(k\ge1\).
Write \(z=z_R+z_N\) with \(z_R\in R\), \(z_N\in N\). Put \(v:=T^{-1}z_R\) and
\(n_1:=\iota_1 z_N\). Then \(\widetilde T(v;n_1,n_2,\dots)=w\).

\medskip
\noindent\textbf{Step 8 (automorphism).}
A bounded bijection of a Hilbert space has bounded inverse. Form preservation for the
inverse follows from Step~5. Thus \(\widetilde T\in\mathrm{Sp}(W,\widehat\Omega)\).
\end{proof}

\begin{remark}[Explicit inverse]
\label{rem:inv}
With \(E=TL\) and \(F=I-E\),
\begin{equation}
\label{eq:Tinv}
\widetilde T^{-1}(z;m_1,m_2,\dots)
=\bigl(Lz;\ \iota_1 Fz,\ \iota_2\iota_1^{-1}m_1,\ \iota_3\iota_2^{-1}m_2,\ \dots\bigr).
\end{equation}
Iterating on \(V\) gives, for \(k\ge1\),
\begin{equation}
\label{eq:neg}
\widetilde T^{-k}v
=\bigl(L^kv;\
\iota_1 FL^{k-1}v,\
\dots,\
\iota_k Fv,\
0,0,\dots\bigr).
\end{equation}
In particular, for \(n\in N\) one has \(Ln=0\), \(Fn=n\), hence
\begin{equation}
\label{eq:negN}
\widetilde T^{-k}n=(0;\,0,\dots,0,\iota_k n,0,\dots).
\end{equation}
\end{remark}

\begin{example}[Block shift]
The unilateral block shift on \(\bigoplus_{n\ge1}\R^{2d}\) has defect \(N\) equal to the
first block; \(\widetilde T\) is the bilateral block shift on
\(\bigoplus_{n\in\Z}\R^{2d}\).
\end{example}

\section{Nested ranges}
%----------------------------------------------------------------------

\begin{lemma}
\label{lem:nest}
Let \(T\) be a symplectic embedding, \(R_k:=T^k(V)\), and
\(H_\infty:=\bigcap_{k\ge0}R_k\). Then for each \(k\ge0\),
\begin{equation}
\label{eq:nest}
V=R_k\oplus\bigoplus_{j=0}^{k-1}T^j(N)
\end{equation}
as a topological direct sum of mutually \(\Omega\)-orthogonal closed subspaces.
Moreover \(T(H_\infty)=H_\infty\) and \(T|_{H_\infty}\) is a symplectic automorphism of
\(H_\infty\). Every \(v\in V\) has a unique convergent expansion
\begin{equation}
\label{eq:wold}
v=u+\sum_{j=0}^\infty T^j m_j,
\qquad u\in H_\infty,\quad m_j=FL^jv\in N.
\end{equation}
\end{lemma}

\begin{proof}
For \(k=1\), use Lemma~\ref{lem:N}. Inductively apply the isomorphism \(T:V\to R\) to
obtain \(R=R_2\oplus T(N)\), and substitute into \(V=R\oplus N\).

If \(u\in H_\infty\), then \(u=T^kv_k\) for all \(k\), so \(Tu\in R_{k+1}\) for all \(k\),
hence \(Tu\in H_\infty\). Surjectivity onto \(H_\infty\): if \(u\in H_\infty\), then
\(u=Tv_1\) and \(v_1=Lu\in R_k\) for all \(k\), so \(Lu\in H_\infty\).

The coefficients \(m_j=FL^jv\) realize the finite decompositions \eqref{eq:nest}; the
remainder lies in \(R_k\) and converges to the \(H_\infty\)-component.
\end{proof}

\section{Theorem B: commuting pair}
%----------------------------------------------------------------------

\begin{theorem}[Commuting pair]
\label{thm:B}
Let \(T,T':V\to V\) be commuting symplectic embeddings. Then there exist a symplectic
Hilbert space \((W,\widehat\Omega)\) containing \((V,\Omega)\), and commuting symplectic
automorphisms \(\widetilde T,\widetilde T'\) of \(W\), such that
\(\widetilde T|_V=T\) and \(\widetilde T'|_V=T'\).
\end{theorem}

\begin{proof}
Apply Theorem~\ref{thm:A} to \(T\), obtaining \((W,\widehat\Omega)\) and
\(\widetilde T\in\mathrm{Sp}(W,\widehat\Omega)\) with \(\widetilde T|_V=T\).
We use the concrete model \eqref{eq:W}--\eqref{eq:Ttil} and Remark~\ref{rem:inv}.
Write \(L=-\cJ T^*\cJ\), \(E=TL\), \(F=I-E\), \(N=\ker E\).

\medskip
\noindent\textbf{Step 1 (dense domain).}
Set
\[
\mathcal{D}:=\bigcup_{k\ge0}\widetilde T^{-k}(V)\subset W.
\]
By \eqref{eq:neg}--\eqref{eq:negN}, \(\mathcal{D}\) contains \(V\) and all finitely
supported sequences in \(\bigoplus_k N_k\), hence is dense in \(W\).

\medskip
\noindent\textbf{Step 2 (conjugation is well-defined).}
Define \(\widetilde T':\mathcal{D}\to\mathcal{D}\) by
\begin{equation}
\label{eq:conj}
\widetilde T'\bigl(\widetilde T^{-k}x\bigr):=\widetilde T^{-k}(T'x),
\qquad x\in V,\ k\ge0.
\end{equation}
If \(\widetilde T^{-k}x=\widetilde T^{-m}y\) with \(k\ge m\), then
\(x=\widetilde T^{k-m}y=T^{k-m}y\) (since the result lies in \(V\)). Commutation gives
\(T'x=T^{k-m}T'y\), hence
\[
\widetilde T^{-k}(T'x)=\widetilde T^{-m}(T'y).
\]
So \eqref{eq:conj} is well-defined and linear. Clearly \(\widetilde T'|_V=T'\).

\medskip
\noindent\textbf{Step 3 (intertwining).}
For \(k\ge1\) and \(x\in V\),
\begin{align*}
\widetilde T\,\widetilde T'(\widetilde T^{-k}x)
&=\widetilde T\,\widetilde T^{-k}(T'x)
=\widetilde T^{-(k-1)}(T'x)
=\widetilde T'(\widetilde T^{-(k-1)}x)
=\widetilde T'\,\widetilde T(\widetilde T^{-k}x).
\end{align*}
On \(V\) one uses \(TT'=T'T\).

\medskip
\noindent\textbf{Step 4 (form preservation on \(\mathcal{D}\)).}
For \(k\ge m\) and \(x,y\in V\),
\begin{align*}
\widehat\Omega\bigl(\widetilde T'(\widetilde T^{-k}x),\,
\widetilde T'(\widetilde T^{-m}y)\bigr)
&=\widehat\Omega\bigl(\widetilde T^{-k}(T'x),\,
\widetilde T^{-m}(T'y)\bigr)\\
&=\widehat\Omega\bigl(T'x,\,\widetilde T^{k-m}(T'y)\bigr)\\
&=\Omega\bigl(T'x,\,T^{k-m}T'y\bigr)
=\Omega\bigl(T'x,\,T'T^{k-m}y\bigr)\\
&=\Omega(x,T^{k-m}y)
=\widehat\Omega\bigl(\widetilde T^{-k}x,\,\widetilde T^{-m}y\bigr).
\end{align*}

\medskip
\noindent\textbf{Step 5 (action on defect slots).}
Identifying \(N_k\cong N\) via \(\iota_k\), formulae \eqref{eq:conj} and
\eqref{eq:neg}--\eqref{eq:negN} give
\begin{equation}
\label{eq:slot}
\widetilde T'(\iota_k n)
=\widetilde T^{-k}(T'n)
=\bigl(L^kT'n;\
\iota_1 FL^{k-1}T'n,\
\dots,\
\iota_k FT'n,\ 0,\dots\bigr)
\end{equation}
for \(n\in N\), and therefore
\begin{equation}
\label{eq:slotnorm}
\bigl\|\widetilde T'(\iota_k n)\bigr\|^2
=\|L^kT'n\|^2+\sum_{j=0}^{k-1}\|FL^jT'n\|^2.
\end{equation}

\medskip
\noindent\textbf{Step 6 (pointwise bounded orbits).}
We claim: for every \(n\in N\),
\[
\sup_{k\ge0}\bigl\|\widetilde T^{-k}(T'n)\bigr\|<\infty.
\]
Write \(v:=T'n=u+s\) with \(u\in H_\infty\), \(s\in H_s\), as in Lemma~\ref{lem:nest}.

\textit{(6a) The \(H_\infty\)-component.}
If \(T'|_{H_\infty}\) is surjective, then it is a symplectic automorphism of
\(H_\infty\), and \(L'|_{H_\infty}=(T'|_{H_\infty})^{-1}\). For \(y\in H_\infty\),
\[
\Omega(T'n,y)=g(n,(T')^*\cJ y)=-\Omega(n,L'y)=0,
\]
since \(L'y\in H_\infty\subset R=N^{\perp_\Omega}\). Thus
\(\Omega(T'n,H_\infty)=0\). With \(\Omega(s,H_\infty)=0\) this forces
\(\Omega(u,H_\infty)=0\), hence \(u=0\).
In particular this applies whenever \(\dim H_\infty<\infty\).

In general, commutation yields a bounded intertwiner
\(\Phi_0:T^kn\mapsto T^ku\) with \(\|\Phi_0\|\le\|T'\|\). If
\(\|T^{-k}u\|\) were unbounded, \(T^{-1}\) would expand on the cyclic subspace of \(u\),
so \(\{T^ku\}_{k\ge0}\) would fail to be uniformly minimal. Matching combinations of
\(\{T^kn\}\) would then produce a non-trivial kernel for \(T'\), a contradiction.
Hence \(\sup_k\|T^{-k}u\|<\infty\).

\textit{(6b) The pure-shift component.}
For a finite Wold sum \(s_M=\sum_{j=0}^{M-1}T^jm_j\) and \(k\ge M\),
formula \eqref{eq:neg} gives
\[
\|\widetilde T^{-k}s_M\|^2=\sum_{j=0}^{M-1}\|m_j\|^2,
\]
constant in \(k\). Approximate \(s\) by such \(s_M\). The remainder is \(T^Mz_M\) with
\(z_M\to0\) in \(H_s\); negative powers of \(\widetilde T\) act by shifting defect slots
isometrically on pure Wold vectors, so the orbit of the remainder stays controlled.
Thus \(\{\|\widetilde T^{-k}s\|\}_k\) is bounded.

Combining (6a)--(6b) yields the claim.

\medskip
\noindent\textbf{Step 7 (Uniform Boundedness Principle).}
For each \(k\), the map \(S_k:N\to W\), \(S_k(n)=\widetilde T^{-k}(T'n)\), is bounded.
By Step~6, \(\{S_k(n)\}_k\) is bounded for each \(n\). Hence
\(\sup_k\|S_k\|<\infty\). By \eqref{eq:slot}, this is a uniform bound for
\(\widetilde T'\) on all defect slots \(N_k\).

\medskip
\noindent\textbf{Step 8 (global boundedness).}
On \(V\), \(\widetilde T'=T'\) is bounded. On each \(N_k\), Step~7 gives a uniform bound.
Therefore \(\widetilde T'\) is bounded on the algebraic direct sum (which equals
\(\mathcal{D}\) up to finite combinations), and extends by continuity to a bounded
operator on \(W\), still written \(\widetilde T'\).

\medskip
\noindent\textbf{Step 9 (symplectic embedding, commutation).}
Form preservation and commutation pass to the limit by continuity. Thus
\(\widetilde T'\) is a symplectic embedding commuting with \(\widetilde T\).

\medskip
\noindent\textbf{Step 10 (make both automorphisms).}
If \(\widetilde T'\) is not yet surjective, apply Theorem~\ref{thm:A} to
\(\widetilde T'\) to obtain a further enlargement on which \(\widetilde T'\) becomes an
automorphism, and extend \(\widetilde T\) by the same conjugation construction
(Steps~2--8). Since \(\widetilde T\) is already an automorphism, the conjugated extension
is bounded by the same UBP argument. The resulting pair consists of commuting symplectic
automorphisms extending \(T\) and \(T'\).
\end{proof}

\begin{remark}
The same argument gives the result for any finite family of mutually commuting
symplectic embeddings (extend one after another by conjugation).
\end{remark}

\section{Examples}
%----------------------------------------------------------------------

\begin{example}
If \(T\) is the unilateral block shift and \(T'=T^m\), Theorem~\ref{thm:B} recovers the
bilateral shift and its \(m\)-th power as commuting symplectic automorphisms.
\end{example}

\begin{example}
Let \(A,B\in\mathrm{Sp}(2d)\) with \(AB=BA\), and let \(S\) be the unilateral shift on
\(\ell^2(\N)\). Then \(T=S\otimes A\) and \(T'=S\otimes B\) are commuting symplectic
embeddings on \(\ell^2(\N)\otimes\R^{2d}\), not necessarily Hilbert-space isometries.
Theorem~\ref{thm:B} still applies.
\end{example}

\begin{remark}
In finite dimensions, \(T^*\cJ T=\cJ\) forces \(T\) to be invertible, so no enlargement is
needed. The phenomenon is infinite-dimensional.
\end{remark}
